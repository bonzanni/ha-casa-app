abi <abi/3.0>,

#include <tunables/global>

profile casa flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),
  capability fowner,
  capability chown,
  capability dac_override,
  capability setuid,
  capability setgid,
  # Casa and s6 run as root and must signal an engagement's processes, which
  # run as that engagement's own uid; without this the kernel refuses (#1382).
  capability kill,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /usr/local/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /run/{,**} rwk,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rwix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Python
  /usr/lib/python3*/** r,
  /usr/local/lib/python3*/** rix,

  # Node.js (Claude Agent SDK)
  /usr/lib/node_modules/** rix,

  # nginx
  /etc/nginx/** rw,
  /var/log/nginx/** rw,
  /var/lib/nginx/** rw,

  # Persistent data and config
  /config/ r,
  /config/** rwk,
  /data/** rwk,
  /opt/casa/** rix,

  # Network. Enforced only because of the abi line at the top of this file;
  # without it the kernel ignores these rules. TCP and UDP stay open to any
  # destination, over IPv4 and IPv6. Socket kinds outside these rules, such as
  # inet raw, packet and netlink dgram, are refused.
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  # glibc getaddrinfo (Python and the claude CLI) reads addresses over NETLINK_ROUTE.
  network netlink raw,
  # Pathname unix sockets fall under the file rules, and socketpairs between
  # Casa's own processes under abstractions/base. This rule is kept as a margin
  # for unix stream use that relies on neither.
  network unix stream,

  # Proc and sys
  /proc/** r,
  /sys/** r,

  # The script runner (docs/architecture/script-runner.md): the one interpreter at this
  # exact path runs under the child profile below. An exact path outranks /opt/casa/** rix.
  /opt/casa/run/python3 cx -> pyrun,

  # The script runner's box. No network rule, so with the abi line every socket family is
  # refused; no x rule, so nothing can be run; no ptrace, no mount, no capability, no write
  # anywhere. It reads the interpreter, its own prelude, the C libraries, the standard library
  # and the stdlib's public data (locale, glibc's charset tables, zoneinfo, the OpenSSL config). Nothing under /data,
  # /config, /etc beyond the files named, /proc beyond a confinement label and its own fd
  # listing, /opt/casa beyond /opt/casa/run. What the profile does not mediate, the prelude's
  # syscall filter refuses. pyrun.<digest> names this text: /opt/casa/scripts/pyrun_digest.py
  # computes the digest over these lines without that one, and child.py and the Dockerfile
  # carry it. Never add a rule for one of the prelude's intentional refusals (/etc/passwd,
  # /proc/1/cmdline, a unix or inet socket): the prelude refuses to run when one is allowed.
  profile pyrun flags=(mediate_deleted) {
    /opt/casa/run/python3 rm,
    /opt/casa/run/child.py r,
    /opt/casa/run/pyrun.954209a51d6d r,
    /etc/ld.so.cache r,
    /usr/lib/@{multiarch}/*.so* rm,
    /usr/lib/python3.[0-9]*/ r,
    /usr/lib/python3.[0-9]*/** rm,
    /usr/lib/locale/** r,
    /usr/lib/@{multiarch}/gconv/gconv-modules r,
    /usr/lib/@{multiarch}/gconv/gconv-modules.cache r,
    /usr/lib/@{multiarch}/gconv/gconv-modules.d/ r,
    /usr/lib/@{multiarch}/gconv/gconv-modules.d/*.conf r,
    /usr/share/zoneinfo/** r,
    /etc/localtime r,
    /etc/ssl/openssl.cnf r,
    @{PROC}/@{pid}/attr/apparmor/current r,
    @{PROC}/@{pid}/attr/current r,
    @{PROC}/@{pid}/fd/ r,
    signal (receive) set=(kill, term),
  }
}

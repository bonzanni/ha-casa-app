abi <abi/3.0>,

#include <tunables/global>

profile casa flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),
  capability fowner,
  capability chown,
  capability dac_override,
  capability setuid,
  capability setgid,
  # Casa and s6 run as root and must signal an engagement's processes, which
  # run as that engagement's own uid; without this the kernel refuses (#1382).
  capability kill,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /usr/local/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /run/{,**} rwk,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rwix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Python
  /usr/lib/python3*/** r,
  /usr/local/lib/python3*/** rix,

  # Node.js (Claude Agent SDK)
  /usr/lib/node_modules/** rix,

  # nginx
  /etc/nginx/** rw,
  /var/log/nginx/** rw,
  /var/lib/nginx/** rw,

  # Persistent data and config
  /config/ r,
  /config/** rwk,
  /data/** rwk,
  /opt/casa/** rix,

  # Network. Enforced only because of the abi line at the top of this file;
  # without it the kernel ignores these rules. TCP and UDP stay open to any
  # destination, over IPv4 and IPv6. Socket kinds outside these rules, such as
  # inet raw, packet and netlink dgram, are refused.
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  # glibc getaddrinfo (Python and the claude CLI) reads addresses over NETLINK_ROUTE.
  network netlink raw,
  # Pathname unix sockets fall under the file rules, and socketpairs between
  # Casa's own processes under abstractions/base. This rule is kept as a margin
  # for unix stream use that relies on neither.
  network unix stream,

  # Proc and sys
  /proc/** r,
  /sys/** r,
}
